cmake_minimum_required(VERSION 3.20)
project(streamcipher_consumer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CONSUME_INSTALLED "Use the installed package" OFF)
get_filename_component(repository "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(CONSUME_INSTALLED)
    find_package(streamcipher 0.1 CONFIG REQUIRED)
else()
    add_subdirectory("${repository}" streamcipher-build EXCLUDE_FROM_ALL)
    foreach(unwanted scipher test_gf bench_aes encrypt_file)
        if(TARGET ${unwanted})
            message(FATAL_ERROR "Embedded build unexpectedly created ${unwanted}")
        endif()
    endforeach()
endif()
if(NOT CMAKE_CXX_STANDARD EQUAL 17)
    message(FATAL_ERROR "Dependency changed the consumer's C++ standard")
endif()
get_target_property(definitions streamcipher::streamcipher INTERFACE_COMPILE_DEFINITIONS)
if(definitions MATCHES "STREAMCIPHER_(SIMD|AESNI)")
    message(FATAL_ERROR "Implementation flags leaked to consumers")
endif()
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE streamcipher::streamcipher)
enable_testing()
add_test(NAME consumer COMMAND consumer)
